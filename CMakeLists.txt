cmake_minimum_required(VERSION 3.16)
project(stock_portfolio_lab C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(portfolio_core src/core.c src/algorithms.c)
target_include_directories(portfolio_core PUBLIC include)
if(MSVC)
  target_compile_options(portfolio_core PRIVATE /W4)
else()
  target_compile_options(portfolio_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(portfolio_lab src/main.c)
add_executable(portfolio_tests tests/tests.c)
add_executable(benchmark_runner src/benchmark.c)
target_link_libraries(portfolio_lab PRIVATE portfolio_core)
target_link_libraries(portfolio_tests PRIVATE portfolio_core)
target_link_libraries(benchmark_runner PRIVATE portfolio_core)
if(NOT MSVC)
  target_link_libraries(portfolio_lab PRIVATE m)
  target_link_libraries(portfolio_tests PRIVATE m)
  target_link_libraries(benchmark_runner PRIVATE m)
endif()
enable_testing()
add_test(NAME portfolio_tests COMMAND portfolio_tests)
